🔹 Backend (Node.js + Express)

import express from "express";
import fetch from "node-fetch"; // if Node < 18, install `node-fetch`

const app = express();
app.use(express.json());

// Replace with your PhonePe sandbox/prod creds
const CLIENT_ID = "YOUR_CLIENT_ID";
const CLIENT_SECRET = "YOUR_CLIENT_SECRET";
const CLIENT_VERSION = "YOUR_CLIENT_VERSION";

// Get Auth Token
async function getAuthToken() {
  const resp = await fetch(
    "https://api-preprod.phonepe.com/apis/pg-sandbox/v1/oauth/token",
    {
      method: "POST",
      headers: { "Content-Type": "application/x-www-form-urlencoded" },
      body: new URLSearchParams({
        client_id: CLIENT_ID,
        client_secret: CLIENT_SECRET,
        client_version: CLIENT_VERSION,
        grant_type: "client_credentials",
      }),
    }
  );
  const data = await resp.json();
  return data.access_token;
}

// Create Order API for your app
app.post("/api/create-order", async (req, res) => {
  try {
    const { amount, orderId } = req.body;
    const authToken = await getAuthToken();

    const resp = await fetch(
      "https://api-preprod.phonepe.com/apis/pg-sandbox/checkout/v2/pay",
      {
        method: "POST",
        headers: {
          "Content-Type": "application/json",
          Authorization: `O-Bearer ${authToken}`,
        },
        body: JSON.stringify({
          merchantOrderId: orderId,
          amount: amount, // in paisa, e.g. ₹10 = 1000
          paymentFlow: {
            type: "PG_CHECKOUT",
            merchantUrls: {
              redirectUrl: "https://yourdomain.com/payment-callback",
            },
          },
        }),
      }
    );

    const data = await resp.json();
    res.json({ redirectUrl: data.redirectUrl });
  } catch (err) {
    console.error(err);
    res.status(500).json({ error: "Failed to create order" });
  }
});

// Webhook endpoint (optional but recommended)
app.post("/api/phonepe-webhook", (req, res) => {
  console.log("Webhook received:", req.body);
  // TODO: update order status in DB
  res.sendStatus(200);
});

app.listen(5000, () => console.log("Server running on port 5000"));

🔹 Expo App (React Native with WebView)

File: PhonePePayment.tsx

import React, { useState, useEffect } from "react";
import { View, ActivityIndicator } from "react-native";
import { WebView } from "react-native-webview";

export default function PhonePePayment({ navigation }) {
  const [redirectUrl, setRedirectUrl] = useState("");

  useEffect(() => {
    const createOrder = async () => {
      try {
        const res = await fetch("http://YOUR_PC_IP:5000/api/create-order", {
          method: "POST",
          headers: { "Content-Type": "application/json" },
          body: JSON.stringify({
            amount: 1000, // ₹10 in paisa
            orderId: "TX" + Date.now(),
          }),
        });
        const data = await res.json();
        setRedirectUrl(data.redirectUrl);
      } catch (err) {
        console.error(err);
      }
    };
    createOrder();
  }, []);

  if (!redirectUrl) {
    return (
      <View style={{ flex: 1, justifyContent: "center" }}>
        <ActivityIndicator size="large" />
      </View>
    );
  }

  return (
    <WebView
      source={{ uri: redirectUrl }}
      onNavigationStateChange={(navState) => {
        if (navState.url.includes("https://yourdomain.com/payment-callback")) {
          navigation.replace("PaymentStatus", { url: navState.url });
        }
      }}
    />
  );
}

🔹 Why does PhonePe need a redirectUrl?

PhonePe’s checkout flow is web-based.

After payment (success/failure/cancel), it always redirects the user to a URL you specify.

This redirect is how PhonePe signals “the flow is done”.

🔹 Options for redirectUrl in a mobile app

Point it to your backend

Example: https://api.mybackend.com/payment-callback

PhonePe will redirect the user there after payment.

You capture the event, verify payment on the backend, then return a simple “Payment Successful” page (which your WebView will load).

From the WebView, you detect this URL and close/redirect inside your app.

Use a deep link / universal link

Example: myapp://payment-callback or https://myapp.page.link/payment-callback

When PhonePe finishes, it redirects to your deep link.

Your app opens directly instead of showing a web page.

Requires you to set up deep linking in Expo (works with bare or managed workflow + linking config).

Hybrid (recommended)

Redirect to backend URL → backend does status verification → backend responds with a minimal web page (success/failure).

Your WebView in the app loads this page, and you can detect completion.

Keeps logic secure and easier to test.
